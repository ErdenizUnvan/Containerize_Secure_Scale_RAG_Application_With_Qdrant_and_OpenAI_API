Create the `nginx-ssl` directory inside `/home/admin` and make sure it contains `certificate.pem` and `privatekey.pem`.

```bash
mkdir nginx-ssl

cd nginx-ssl

openssl req -x509 -newkey rsa:4096 -nodes -out certificate.pem -keyout privatekey.pem -days 3650
```

`privatekey.pem` and `certificate.pem` are located inside the `/home/admin/nginx-ssl` directory.

```bash
cd ..
```

Create the `lb` directory inside `/home/admin/`. The `nginx.conf` file is located inside the `lb` directory:

```bash
mkdir lb

cd lb

cat > nginx.conf
```

```nginx
events {}

http {
    upstream myapp {
        server 172.20.0.110:8443;
        server 172.20.0.111:8443;
        server 172.20.0.112:8443;
        server 172.20.0.113:8443;
    }

    server {
        listen 8443 ssl;

        server_name localhost;

        ssl_certificate     /etc/nginx/ssl/certificate.pem;
        ssl_certificate_key /etc/nginx/ssl/privatekey.pem;

        location / {
            proxy_pass https://myapp;
            proxy_ssl_verify off;
            proxy_set_header Host $host;
        }
    }
}
```
